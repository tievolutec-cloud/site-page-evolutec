  function speakSequence(parts,onDone){
    if(!("speechSynthesis" in window)||!bestVoice()){onDone?.();return}
    cancelSpeech();
    const token=speechToken;
    const queue=(Array.isArray(parts)?parts:[parts]).map(p=>typeof p==="string"?{text:p}:p).filter(p=>p&&p.text);
    let i=0;
    const next=()=>{
      if(token!==speechToken)return;
      if(i>=queue.length){onDone?.();return}
      const part=queue[i++];
      speechTimer=setTimeout(()=>{
        if(token!==speechToken)return;
        const u=utteranceFor(part.text,{rate:part.rate??.98,pitch:part.pitch??1.02});
        if(!u){onDone?.();return}
        let finished=false;
        const finish=()=>{if(finished)return;finished=true;setTimeout(next,part.pauseAfter??180)};
        u.onend=finish;u.onerror=finish;
        speechSynthesis.speak(u);
      },part.delay??60);
    };
    next();
  }
