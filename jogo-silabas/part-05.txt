  function buildChoose(b){
    if(session.lock)return;const {w,selected}=session.current;const pos=selected.length;state.attempts[session.id]=(state.attempts[session.id]||0)+1;
    if(b.dataset.s===w.syll[pos]){selected.push(b.dataset.s);b.disabled=true;const slot=document.querySelectorAll(".slot")[pos];slot.textContent=b.dataset.s;slot.classList.add("filled");beep(true);speakSyllable(b.dataset.s);if(selected.length===w.syll.length){session.lock=true;session.correct++;state.correct[session.id]=(state.correct[session.id]||0)+1;state.stars+=3;feedback("Você montou "+w.word+"! 🌟");confetti();save();updateStats();showNext();}}
    else{session.mistakes++;session.lock=true;beep(false);b.classList.add("wrong");feedback(`Vamos montar devagar: ${w.syll.join(" • ")}.`,`try`);setTimeout(()=>b.classList.remove("wrong"),650);const needed=w.syll[pos];setTimeout(()=>speakSequence([{text:"Essa sílaba ainda não.",rate:.98,pauseAfter:350},...w.syll.map((syl,i)=>({text:syllableSound(syl),rate:.93,pauseAfter:i===w.syll.length-1?450:560})),{text:`Agora precisamos da sílaba`,rate:.98,pauseAfter:260},{text:syllableSound(needed),rate:.92,pauseAfter:420},{text:"Tente novamente.",rate:.98}],()=>{session.lock=false}),300);save();}
  }
